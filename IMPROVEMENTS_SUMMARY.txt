BASIC USAGE:
  ./parse -file myfile.py -use_tags_query -tags_query_dir queries

FIND VARIABLES:
  ./parse -file myfile.py -use_tags_query -tags_query_dir queries \
    | grep "definition.variable"

FIND ENUMS:
  ./parse -file myfile.java -use_tags_query -tags_query_dir queries \
    | grep "definition.enum"

ANALYZE EXCEPTIONS:
  ./parse -file myfile.ts -use_tags_query -tags_query_dir queries \
    | grep "exception_handler"

TEST & VALIDATE:
  ./test.sh python              # Test Python only
  ./validate_queries.sh go      # Validate Go queries

BATCH PROCESSING:
  for file in $(find . -name "*.java"); do
    ./parse -file "$file" -use_tags_query -tags_query_dir queries \
      | grep -c "exception"
  done
